cmake_minimum_required(VERSION 3.16)
project(eskf LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ESKF_BUILD_ROS "Build the ROS 2 adapter" ON)
find_package(Eigen3 REQUIRED)
add_library(eskf_core src/eskf.cpp)
set_target_properties(eskf_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(eskf_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(eskf_core PUBLIC Eigen3::Eigen)
option(ESKF_WITH_GTSAM "Build the host Forster preintegration/fixed-lag backend" ON)
if(ESKF_WITH_GTSAM)
  find_package(GTSAM 4.2.0 EXACT REQUIRED)
  find_package(GTSAM_UNSTABLE REQUIRED)
  add_library(eskf_gtsam src/gtsam_navigation.cpp)
  set_target_properties(eskf_gtsam PROPERTIES POSITION_INDEPENDENT_CODE ON)
  target_link_libraries(eskf_gtsam PUBLIC eskf_core gtsam gtsam_unstable)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(eskf_core_tests test/core_tests.cpp)
  target_link_libraries(eskf_core_tests PRIVATE eskf_core)
  add_test(NAME eskf_core_tests COMMAND eskf_core_tests)
  if(ESKF_WITH_GTSAM)
    add_executable(eskf_gtsam_tests test/gtsam_tests.cpp)
    target_link_libraries(eskf_gtsam_tests PRIVATE eskf_gtsam)
    add_test(NAME eskf_gtsam_tests COMMAND eskf_gtsam_tests)
  endif()
endif()
if(ESKF_BUILD_ROS)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(rclcpp_components REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(std_srvs REQUIRED)
  find_package(nav_msgs REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(tf2_ros REQUIRED)
  find_package(tf2_eigen REQUIRED)
  find_package(spdlog REQUIRED)
  add_library(eskf_component SHARED src/eskf_ros.cpp)
  target_link_libraries(eskf_component PUBLIC eskf_core spdlog::spdlog)
  if(ESKF_WITH_GTSAM)
    target_link_libraries(eskf_component PUBLIC eskf_gtsam)
    target_compile_definitions(eskf_component PRIVATE ESKF_WITH_GTSAM)
    install(TARGETS eskf_gtsam EXPORT export_eskf ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
    ament_export_dependencies(GTSAM GTSAM_UNSTABLE)
  endif()
  ament_target_dependencies(eskf_component PUBLIC
    rclcpp rclcpp_components sensor_msgs std_msgs std_srvs
    nav_msgs geometry_msgs tf2_ros tf2_eigen)
  rclcpp_components_register_node(eskf_component PLUGIN "ESKFNode" EXECUTABLE eskf_node)
  install(TARGETS eskf_core eskf_component EXPORT export_eskf ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
  install(DIRECTORY include/ DESTINATION include)
  install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
  ament_export_targets(export_eskf HAS_LIBRARY_TARGET)
  ament_export_dependencies(Eigen3 rclcpp sensor_msgs std_msgs std_srvs nav_msgs geometry_msgs tf2_ros tf2_eigen spdlog)
  if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    # Run after installation with its setup.bash sourced (see README).
    install(PROGRAMS test/ros_contract_test.py DESTINATION lib/${PROJECT_NAME})
  endif()
  ament_package()
endif()
